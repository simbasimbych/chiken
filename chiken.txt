<body>
<canvas id="gameCanvas" width="500" height="800"></canvas>
<div id="overlay">
  <div>
    <h2>Chicken Cross</h2>
    <p>Swipe or use arrows/WASD to move</p>
    <button onclick="startGame()">Start Game</button>
  </div>
</div>
<script>
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
const overlay = document.getElementById('overlay');

let chicken, cars, corns, score, lives, gameSpeed, gameStarted, gameOver, highScore = 0;

function resetGame() {
  chicken = { x: 225, y: 700, size: 30, dx: 0, dy: 0 };
  cars = [];
  corns = [];
  score = 0;
  lives = 3;
  gameSpeed = 2;
  gameStarted = true;
  gameOver = false;
}

function drawChicken() {
  const { x, y, size } = chicken;
  ctx.save();
  ctx.translate(x + size / 2, y + size / 2);

  ctx.fillStyle = "#FFD700";
  ctx.beginPath();
  ctx.ellipse(0, 0, size / 2 + 5, size / 2, 0, 0, Math.PI * 2);
  ctx.fill();

  ctx.fillStyle = "#FFC700";
  ctx.beginPath();
  ctx.ellipse(-5, 5, 6, 10, 0, 0, Math.PI * 2);
  ctx.fill();

  ctx.fillStyle = "black";
  ctx.beginPath();
  ctx.arc(10, -5, 3, 0, Math.PI * 2);
  ctx.fill();

  ctx.fillStyle = "orange";
  ctx.beginPath();
  ctx.moveTo(18, 0);
  ctx.lineTo(25, 3);
  ctx.lineTo(18, 6);
  ctx.closePath();
  ctx.fill();

  ctx.fillStyle = "red";
  ctx.beginPath();
  ctx.arc(-5, -15, 4, 0, Math.PI * 2);
  ctx.arc(0, -17, 4, 0, Math.PI * 2);
  ctx.arc(5, -15, 4, 0, Math.PI * 2);
  ctx.fill();

  ctx.restore();
}

function drawCorn(corn) {
  ctx.save();
  ctx.translate(corn.x, corn.y);
  ctx.fillStyle = "green";
  ctx.beginPath();
  ctx.ellipse(5, 10, 6, 10, 0, 0, Math.PI * 2);
  ctx.fill();
  ctx.fillStyle = "#FFD700";
  ctx.beginPath();
  ctx.ellipse(5, 8, 4, 8, 0, 0, Math.PI * 2);
  ctx.fill();
  ctx.restore();
}

function drawCar(car) {
  ctx.fillStyle = car.color;
  ctx.fillRect(car.x, car.y, car.width, car.height);
  ctx.fillStyle = "black";
  ctx.fillRect(car.x + 5, car.y - 5, 10, 5);
  ctx.fillRect(car.x + car.width - 15, car.y - 5, 10, 5);
  ctx.fillRect(car.x + 5, car.y + car.height, 10, 5);
  ctx.fillRect(car.x + car.width - 15, car.y + car.height, 10, 5);
}

function drawRoadLanes() {
  for (let i = 1; i < 8; i++) {
    ctx.strokeStyle = "white";
    ctx.lineWidth = 2;
    ctx.setLineDash([10, 15]);
    ctx.beginPath();
    ctx.moveTo(0, i * 100);
    ctx.lineTo(canvas.width, i * 100);
    ctx.stroke();
  }
  ctx.setLineDash([]);
}

function drawUI() {
  ctx.fillStyle = "white";
  ctx.font = "18px sans-serif";
  ctx.fillText(`Score: ${score}`, 10, 25);
  ctx.fillText(`Lives: ${lives}`, canvas.width - 90, 25);
  ctx.fillText(`High Score: ${highScore}`, canvas.width / 2 - 70, 25);
}

function spawnCar() {
  const direction = Math.random() < 0.5 ? 1 : -1;
  const width = 60 + Math.random() * 40;
  const y = (Math.floor(Math.random() * 6) + 1) * 100 + 20;
  const baseSpeed = 2 + gameSpeed * 0.5 + score * 0.05;
  cars.push({
    x: direction === 1 ? -width : canvas.width + width,
    y,
    width,
    height: 30,
    speed: direction * baseSpeed,
    color: `hsl(${Math.random()*360}, 70%, 50%)`
  });
  const delay = Math.max(500, 1500 - score * 10);
  setTimeout(spawnCar, delay);
}

function spawnCorn() {
  const corn = {
    x: Math.random() * (canvas.width - 20),
    y: Math.random() * (canvas.height - 200) + 100,
    timestamp: Date.now()
  };
  corns.push(corn);
  setTimeout(() => {
    const index = corns.indexOf(corn);
    if (index > -1 && Date.now() - corn.timestamp >= 5000) {
      corns.splice(index, 1);
    }
  }, 5000);
  setTimeout(spawnCorn, 4000);
}

function checkCollision(a, b) {
  return a.x < b.x + b.width && a.x + a.size > b.x && a.y < b.y + b.height && a.y + a.size > b.y;
}

function update() {
  if (!gameStarted) return;
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  drawRoadLanes();

  chicken.x += chicken.dx;
  chicken.y += chicken.dy;
  chicken.x = Math.max(0, Math.min(canvas.width - chicken.size, chicken.x));
  chicken.y = Math.max(0, Math.min(canvas.height - chicken.size, chicken.y));

  drawChicken();

  for (let car of cars) {
    car.x += car.speed;
    drawCar(car);
    if (checkCollision(chicken, car)) {
      lives--;
      chicken.x = 225;
      chicken.y = 700;
      if (lives <= 0) return endGame();
    }
  }
  cars = cars.filter(c => c.x + c.width > 0 && c.x < canvas.width);

  for (let corn of corns) drawCorn(corn);
  for (let i = corns.length - 1; i >= 0; i--) {
    if (Math.hypot(chicken.x - corns[i].x, chicken.y - corns[i].y) < 20) {
      score += 5;
      corns.splice(i, 1);
    }
  }

  if (chicken.y < 50) {
    score++;
    chicken.x = 225;
    chicken.y = 700;
    gameSpeed += 0.1;
  }

  drawUI();
  requestAnimationFrame(update);
}

function endGame() {
  gameStarted = false;
  if (score > highScore) highScore = score;
  overlay.innerHTML = `<div><h2>Game Over</h2><p>Final Score: ${score}</p><p>High Score: ${highScore}</p><button onclick="startGame()">Restart</button></div>`;
  overlay.style.display = 'flex';
}

function startGame() {
  resetGame();
  overlay.style.display = 'none';
  spawnCar();
  spawnCorn();
  update();
}

let touchStartX = 0, touchStartY = 0;
canvas.addEventListener('touchstart', e => {
  const touch = e.touches[0];
  touchStartX = touch.clientX;
  touchStartY = touch.clientY;
});
canvas.addEventListener('touchend', e => {
  const dx = e.changedTouches[0].clientX - touchStartX;
  const dy = e.changedTouches[0].clientY - touchStartY;
  if (Math.abs(dx) > Math.abs(dy)) {
    chicken.dx = dx > 0 ? 5 : -5;
    chicken.dy = 0;
  } else {
    chicken.dy = dy > 0 ? 5 : -5;
    chicken.dx = 0;
  }
  setTimeout(() => { chicken.dx = 0; chicken.dy = 0; }, 150);
});

window.addEventListener('keydown', e => {
  if (e.key === 'ArrowLeft' || e.key === 'a') chicken.dx = -5;
  if (e.key === 'ArrowRight' || e.key === 'd') chicken.dx = 5;
  if (e.key === 'ArrowUp' || e.key === 'w') chicken.dy = -5;
  if (e.key === 'ArrowDown' || e.key === 's') chicken.dy = 5;
});
window.addEventListener('keyup', e => {
  if (["ArrowLeft","ArrowRight","a","d"].includes(e.key)) chicken.dx = 0;
  if (["ArrowUp","ArrowDown","w","s"].includes(e.key)) chicken.dy = 0;
});
</script>
